async function proxy(server,profile){
 const sockets=new Set();let origin=null;
 const metrics={connections:0,upBytes:0,downBytes:0,maxQueuedBytes:0,segments:0,serverFrames:[]};
 function elapsed(){return origin===null?0:performance.now()-origin}
 function transport(source,destination,direction){
  const queue=[];let timer=null,queued=0,lastDue=0;let stopped=false,ended=false,waitingDrain=false;
  const stalled=()=>{const t=elapsed();return(profile==='duplex-stall'||profile===`${direction==='up'?'uplink':'downlink'}-stall`)&&t>=18000&&t<34000};
  const clear=()=>{stopped=true;if(timer)clearTimeout(timer);queue.length=0;queued=0};source.once('close',()=>{if(!source.readableEnded)clear()});destination.once('close',clear);source.once('end',()=>{ended=true;if(!queue.length)destination.end()});
  function pump(){timer=null;if(stopped||waitingDrain)return;if(!queue.length){if(ended)destination.end();return}const item=queue[0];const delay=Math.max(item.due-performance.now(),stalled()?origin+34000-performance.now():0);if(delay>0){timer=setTimeout(pump,delay);return}queue.shift();queued-=item.data.length;if(queued<524288&&source.isPaused())source.resume();const ok=destination.write(item.data);if(ok)pump();else{waitingDrain=true;destination.once('drain',()=>{waitingDrain=false;pump()})}}
  let observed=Buffer.alloc(0),handshake=false;
  function observe(data){
   if(direction!=='down')return;
   observed=Buffer.concat([observed,data]);
   if(!handshake){const end=observed.indexOf('\r\n\r\n');if(end<0)return;observed=observed.subarray(end+4);handshake=true}
   while(observed.length>=2){let n=observed[1]&127,h=2;if(n===126){if(observed.length<4)return;n=observed.readUInt16BE(2);h=4}else if(n===127){if(observed.length<10)return;n=Number(observed.readBigUInt64BE(2));h=10}if(observed[1]&128)h+=4;if(observed.length<h+n)return;
    const op=observed[0]&15,compressed=Boolean(observed[0]&64);if(op===1&&!compressed){try{const m=JSON.parse(observed.subarray(h,h+n).toString());if(m.type!=='terminal-output'&&m.type!=='tmux-copy-mode-status')metrics.serverFrames.push({t:Math.round(elapsed()),type:m.type,seq:m.seq,bytes:n})}catch{}}
    observed=observed.subarray(h+n);
   }
  }
  source.on('data',data=>{observe(data);
   const t=elapsed();const shaped=profile!=='baseline';
   // Fixed time-bucket jitter, independent of the builds' TCP chunk boundaries.
   const jitter=profile==='jitter'?Math.round((1+Math.sin(Math.floor(t/250)*1.73+(direction==='up'?0:1)))*175):0;
   const latency=shaped?150+jitter:0;const rate=profile==='rate128'?131072:profile==='rate256'?262144:1e9;
   const paused=(profile==='duplex-stall'||profile===`${direction==='up'?'uplink':'downlink'}-stall`)&&t>=18000&&t<34000;
   for(let offset=0;offset<data.length;offset+=4096){const chunk=data.subarray(offset,offset+4096);let due=Math.max(lastDue,performance.now()+latency,paused?origin+34000:0)+chunk.length/rate*1000;lastDue=due;queue.push({data:chunk,due});queued+=chunk.length;metrics[direction==='up'?'upBytes':'downBytes']+=chunk.length;metrics.segments++;metrics.maxQueuedBytes=Math.max(metrics.maxQueuedBytes,queued)}
   if(queued>1048576)source.pause();
   if(!timer)pump();
  });
 }
 const listener=net.createServer(client=>{if(origin===null)origin=performance.now();metrics.connections++;const upstream=net.connect(server.port,'127.0.0.1');sockets.add(client);sockets.add(upstream);for(const s of [client,upstream]){s.setNoDelay(true);s.on('error',()=>{client.destroy();upstream.destroy()});s.on('close',()=>sockets.delete(s))}transport(client,upstream,'up');transport(upstream,client,'down')});
 await new Promise(r=>listener.listen(0,'127.0.0.1',r));const result={port:listener.address().port,metrics,close(){for(const s of sockets)s.destroy();listener.close()}};proxies.push(result);return result;
}
