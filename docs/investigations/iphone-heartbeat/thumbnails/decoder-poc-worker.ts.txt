import { PNG } from 'pngjs';
import jpeg from 'jpeg-js';
self.onmessage=({data: input})=>{
 try {
 const bytes=Buffer.from(input.bytes);
 const start=performance.now();
 const decoded=input.format==='png'?PNG.sync.read(bytes):jpeg.decode(bytes,{useTArray:true,maxResolutionInMP:4,maxMemoryUsageInMB:96,tolerantDecoding:false});
 const width=Math.min(112,decoded.width),height=Math.max(1,Math.round(decoded.height*width/decoded.width));
 const out=Buffer.alloc(width*height*4);
 for(let y=0;y<height;y++)for(let x=0;x<width;x++){
  const from=(Math.floor(y*decoded.height/height)*decoded.width+Math.floor(x*decoded.width/width))*4;
  const to=(y*width+x)*4;
  for(let c=0;c<4;c++)out[to+c]=decoded.data[from+c];
 }
 const result=PNG.sync.write({width,height,data:out});
 self.postMessage({width,height,bytes:result.length,elapsedMs:performance.now()-start});
 }catch(error){self.postMessage({error:String(error)})}
};
